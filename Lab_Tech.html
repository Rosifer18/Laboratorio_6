// MÓDULO 1: UTILIDADES (Equivalente a utilidades.py)
Funcion resultado ← limpiar_texto(texto)
    Definir inicio, pos_fin Como Entero
    Definir resultado Como Cadena
    inicio ← 1
    pos_fin ← Longitud(texto)
    
    Mientras inicio ≤ pos_fin Y Subcadena(texto, inicio, inicio) = " " Hacer
        inicio ← inicio + 1
    FinMientras
    
    Mientras pos_fin ≥ inicio Y Subcadena(texto, pos_fin, pos_fin) = " " Hacer
        pos_fin ← pos_fin - 1
    FinMientras
    
    Si inicio > pos_fin Entonces
        resultado ← ""
    Sino
        resultado ← Subcadena(texto, inicio, pos_fin)
    FinSi
FinFuncion

Funcion es_valido ← validar_texto_no_vacio(texto)
    Definir es_valido Como Logico
    es_valido ← Longitud(limpiar_texto(texto)) > 0
FinFuncion

Funcion es_valido ← validar_fecha(fecha)
    Definir es_valido Como Logico
    Si Longitud(fecha) = 10 Y Subcadena(fecha, 3, 3) = "/" Y Subcadena(fecha, 6, 6) = "/" Entonces
        es_valido ← Verdadero
    Sino
        es_valido ← Falso
    FinSi
FinFuncion

Funcion entrada ← leer_texto(mensaje)
    Definir entrada Como Cadena
    Repetir
        Escribir Sin Saltar mensaje
        Leer entrada
        entrada ← limpiar_texto(entrada)
        Si NO validar_texto_no_vacio(entrada) Entonces
            Escribir "Error: el campo no puede estar vacío."
        FinSi
    Hasta Que validar_texto_no_vacio(entrada)
FinFuncion

Funcion fecha ← leer_fecha(mensaje)
    Definir fecha Como Cadena
    Definir valida Como Logico
    valida ← Falso
    Repetir
        Escribir Sin Saltar mensaje
        Leer fecha
        fecha ← limpiar_texto(fecha)
        Si validar_texto_no_vacio(fecha) Y validar_fecha(fecha) Entonces
            valida ← Verdadero
        Sino
            Escribir "Error: fecha inválida. Use el formato dd/mm/aaaa (ej. 17/11/2026)."
        FinSi
    Hasta Que valida
FinFuncion

Funcion cat_elegida ← seleccionar_categoria(categorias)
    Definir i, opcion Como Entero
    Definir cat_elegida Como Cadena
    Escribir ""
    Escribir "Categorías disponibles:"
    Para i ← 1 Hasta 5 Hacer
        Escribir "  ", i, ". ", categorias[i]
    FinPara
    
    Repetir
        Escribir Sin Saltar "Seleccione el número de categoría (1-5): "
        Leer opcion
        Si opcion < 1 O opcion > 5 Entonces
            Escribir "Error: seleccione una opción válida de la lista."
        FinSi
    Hasta Que opcion ≥ 1 Y opcion ≤ 5
    
    cat_elegida ← categorias[opcion]
FinFuncion

Funcion indice ← buscar_equipo_por_codigo(eq_codigo, total_equipos, codigo_buscar)
    Definir i, indice Como Entero
    Definir cod_norm Como Cadena
    cod_norm ← Mayusculas(limpiar_texto(codigo_buscar))
    indice ← 0
    
    Para i ← 1 Hasta total_equipos Hacer
        Si Mayusculas(eq_codigo[i]) = cod_norm Entonces
            indice ← i
        FinSi
    FinPara
FinFuncion

// MÓDULO 2: OPERACIONES Y MENÚ PRINCIPAL (Equivalente a main.py)
// 1. Registrar equipo disponible
SubProceso registrar_equipo(eq_codigo, eq_nombre, eq_categoria, eq_estado, total_equipos Por Referencia, categorias)
    Definir codigo, nombre, categoria Como Cadena
    Escribir ""
    Escribir "------- REGISTRAR NUEVO EQUIPO -------"
    codigo ← Mayusculas(leer_texto("Código del equipo (ej. LAP-01): "))
    
    Si buscar_equipo_por_codigo(eq_codigo, total_equipos, codigo) ≠ 0 Entonces
        Escribir "Error: ya existe un equipo registrado con el código ", codigo
    Sino
        nombre ← leer_texto("Nombre o descripción del equipo: ")
        categoria ← seleccionar_categoria(categorias)
        
        total_equipos ← total_equipos + 1
        eq_codigo[total_equipos] ← codigo
        eq_nombre[total_equipos] ← nombre
        eq_categoria[total_equipos] ← categoria
        eq_estado[total_equipos] ← "Disponible"
        
        Escribir ""
        Escribir "Equipo ", nombre, " (", codigo, ") registrado correctamente como Disponible."
    FinSi
FinSubProceso

// 2. Registrar un préstamo
SubProceso registrar_prestamo(eq_codigo, eq_nombre, eq_estado, total_equipos, pr_codigo, pr_estudiante, pr_fecha_p, pr_fecha_d, pr_activo, total_prestamos Por Referencia)
    Definir codigo, estudiante, fecha Como Cadena
    Definir idx Como Entero
    Escribir ""
    Escribir "------- REGISTRAR PRÉSTAMO DE EQUIPO -------"
    
    Si total_equipos = 0 Entonces
        Escribir "No hay equipos registrados en el laboratorio."
    Sino
        codigo ← Mayusculas(leer_texto("Ingrese el código del equipo a prestar: "))
        idx ← buscar_equipo_por_codigo(eq_codigo, total_equipos, codigo)
        
        Si idx = 0 Entonces
            Escribir "Error: el código ", codigo, " no existe en el inventario."
        Sino

            Si eq_estado[idx] ≠ "Disponible" Entonces
                Escribir "Error: el equipo ", eq_nombre[idx], " (", codigo, ") está marcado como No disponible."
            Sino
                estudiante ← leer_texto("Nombre completo del estudiante: ")
                fecha ← leer_fecha("Fecha de préstamo (dd/mm/aaaa): ")
                
                eq_estado[idx] ← "No disponible"
                total_prestamos ← total_prestamos + 1
                pr_codigo[total_prestamos] ← eq_codigo[idx]
                pr_estudiante[total_prestamos] ← estudiante
                pr_fecha_p[total_prestamos] ← fecha
                pr_fecha_d[total_prestamos] ← ""
                pr_activo[total_prestamos] ← Verdadero
                
                Escribir ""
                Escribir "Préstamo exitoso: ", eq_nombre[idx], " asignado a ", estudiante, " (", fecha, ")."
            FinSi
        FinSi
    FinSi
FinSubProceso

// 3. Devolver equipo
SubProceso devolver_equipo(eq_codigo, eq_nombre, eq_estado, total_equipos, pr_codigo, pr_fecha_d, pr_activo, total_prestamos)
    Definir codigo, fecha_dev Como Cadena
    Definir idx, i Como Entero
    Escribir ""
    Escribir "------- DEVOLVER EQUIPO -------"
    
    Si total_equipos = 0 Entonces
        Escribir "No hay equipos registrados."
    Sino
        codigo ← Mayusculas(leer_texto("Ingrese el código del equipo a devolver: "))
        idx ← buscar_equipo_por_codigo(eq_codigo, total_equipos, codigo)
        
        Si idx = 0 Entonces
            Escribir "Error: el código ", codigo, " no existe en el inventario."
        Sino
            Si eq_estado[idx] = "Disponible" Entonces
                Escribir "Aviso: el equipo ", eq_nombre[idx], " (", codigo, ") ya figura como Disponible."
            Sino
                fecha_dev ← leer_fecha("Fecha de devolución (dd/mm/aaaa): ")
                eq_estado[idx] ← "Disponible"
            
                Para i ← 1 Hasta total_prestamos Hacer
                    Si pr_codigo[i] = eq_codigo[idx] Y pr_activo[i] = Verdadero Entonces
                        pr_activo[i] ← Falso
                        pr_fecha_d[i] ← fecha_dev
                    FinSi
                FinPara
                
                Escribir ""
                Escribir "Devolución registrada: ", eq_nombre[idx], " (", codigo, ") vuelve a estar Disponible."
            FinSi
        FinSi
    FinSi
FinSubProceso

// 4. Listar disponibilidad
SubProceso listar_disponibilidad(eq_codigo, eq_nombre, eq_categoria, eq_estado, total_equipos)
    Definir i, disponibles Como Entero
    Escribir ""
    Escribir "------- INVENTARIO Y DISPONIBILIDAD LABTECH -------"
    
    Si total_equipos = 0 Entonces
        Escribir "No hay equipos registrados todavía."
    Sino
        disponibles ← 0
        Escribir "CÓDIGO   | NOMBRE               | CATEGORÍA     | ESTADO"
        Escribir "----------------------------------------------------------------"
        Para i ← 1 Hasta total_equipos Hacer
            Escribir eq_codigo[i], "   | ", eq_nombre[i], "   | ", eq_categoria[i], "   | ", eq_estado[i]
            Si eq_estado[i] = "Disponible" Entonces
                disponibles ← disponibles + 1
            FinSi
        FinPara
        Escribir "----------------------------------------------------------------"
        Escribir "Total equipos: ", total_equipos, " | Disponibles: ", disponibles, " | Prestados: ", total_equipos - disponibles
    FinSi
FinSubProceso

// 5. Consultar cantidad de equipos disponibles por categoría
SubProceso consultar_disponibles_por_categoria(eq_categoria, eq_estado, total_equipos, categorias)
    Definir c, i, cantidad Como Entero
    Escribir ""
    Escribir "------- DISPONIBILIDAD POR CATEGORÍA -------"
    
    Si total_equipos = 0 Entonces
        Escribir "No hay equipos registrados."
    Sino
        Para c ← 1 Hasta 5 Hacer
            cantidad ← 0
            Para i ← 1 Hasta total_equipos Hacer
                Si eq_categoria[i] = categorias[c] Y eq_estado[i] = "Disponible" Entonces
                    cantidad ← cantidad + 1
                FinSi
            FinPara
            Escribir "- ", categorias[c], ": ", cantidad, " disponible(s)"
        FinPara
    FinSi
FinSubProceso

// 6. Mostrar menú principal
SubProceso mostrar_menu
    Escribir ""
    Escribir "========== SISTEMA DE PRÉSTAMOS LABTECH =========="
    Escribir "1. Registrar equipo"
    Escribir "2. Registrar préstamo"
    Escribir "3. Devolver equipo"
    Escribir "4. Listar disponibilidad general"
    Escribir "5. Consultar disponibles por categoría"
    Escribir "6. Salir"
FinSubProceso


// ALGORITMO PRINCIPAL (Punto de entrada main)
Algoritmo SistemaPrestamosLabTech
    // Arreglos paralelos para Equipos (capacidad máxima 100)
    Dimension eq_codigo[100], eq_nombre[100], eq_categoria[100], eq_estado[100]
    // Arreglos paralelos para Préstamos (capacidad máxima 200)
    Dimension pr_codigo[200], pr_estudiante[200], pr_fecha_p[200], pr_fecha_d[200], pr_activo[200]
    // Arreglo de categorías válidas
    Dimension categorias[5]
    
    Definir eq_codigo, eq_nombre, eq_categoria, eq_estado Como Cadena
    Definir pr_codigo, pr_estudiante, pr_fecha_p, pr_fecha_d, categorias Como Cadena
    Definir opcion Como Entero
    Definir pr_activo Como Logico
    Definir total_equipos, total_prestamos Como Entero
    
    categorias[1] ← "Laptop"
    categorias[2] ← "Adaptador"
    categorias[3] ← "Mouse"
    categorias[4] ← "Cable HDMI"
    categorias[5] ← "Kit básico"
    
    total_equipos ← 0
    total_prestamos ← 0
    
    Escribir "Sistema LabTech iniciado correctamente."
    
    Repetir
        mostrar_menu
        
        Escribir Sin Saltar "Seleccione una opción (1-6): "
        Leer opcion
        
        Si opcion < 1 O opcion > 6 Entonces
            Escribir "Error: opción inválida. Elija un número del 1 al 6."
        Sino
            Segun opcion Hacer
                1:
                    registrar_equipo(eq_codigo, eq_nombre, eq_categoria, eq_estado, total_equipos, categorias)
                    
                2:
                    registrar_prestamo(eq_codigo, eq_nombre, eq_estado, total_equipos, pr_codigo, pr_estudiante, pr_fecha_p, pr_fecha_d, pr_activo, total_prestamos)
                    
                3:
                    devolver_equipo(eq_codigo, eq_nombre, eq_estado, total_equipos, pr_codigo, pr_fecha_d, pr_activo, total_prestamos)
                    
                4:
                    listar_disponibilidad(eq_codigo, eq_nombre, eq_categoria, eq_estado, total_equipos)
                    
                5:
                    consultar_disponibles_por_categoria(eq_categoria, eq_estado, total_equipos, categorias)
                    
                6:
                    Escribir "Cerrando sistema LabTech... Hasta pronto."
                    
            FinSegun
        FinSi
        
    Hasta Que opcion = 6
FinAlgoritmo
